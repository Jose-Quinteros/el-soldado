function $(selector) {
    return document.querySelector(selector);
}

function $$(selector) {
    return [...document.querySelectorAll(selector)];
}

function initials(name) {

    return name
        .split(/\s+/)
        .map(word => word[0])
        .join("")
        .slice(0, 2)
        .toUpperCase();
}

function stars(score) {

    const count = Math.max(
        1,
        Math.min(5, Math.round(score / 20))
    );

    return "★".repeat(count) +
           "☆".repeat(5 - count);
}


/* ============================================================
   SCORE
   ============================================================ */

function calcularScore(ia) {

    return Math.round(

        (ia.potencia || 0) * SCORE_WEIGHTS.potencia +

        (ia.calidad || 0) * SCORE_WEIGHTS.calidad +

        (ia.versatilidad || 0) * SCORE_WEIGHTS.versatilidad +

        (ia.popularidad || 0) * SCORE_WEIGHTS.popularidad +

        (ia.innovacion || 0) * SCORE_WEIGHTS.innovacion +

        (ia.facilidad || 0) * SCORE_WEIGHTS.facilidad

    );
}

function loadCustomIA() {

    try {

        return JSON.parse(
            localStorage.getItem(STORAGE.custom) || "[]"
        );

    } catch (error) {

        console.error(
            "No se pudieron cargar las IA personalizadas.",
            error
        );

        return [];
    }
}


function saveCustomIA(data) {

    localStorage.setItem(
        STORAGE.custom,
        JSON.stringify(data)
    );
}


function obtenerTodasLasIA() {

    const custom = loadCustomIA();

    return [
        ...iaDatabase,
        ...custom
    ].map(ia => ({
        ...ia,
        score: calcularScore(ia)
    }));
}

function cargarFavoritos() {

    try {

        return JSON.parse(
            localStorage.getItem(STORAGE.favorites) || "[]"
        );

    } catch {

        return [];
    }
}


function guardarFavoritos(favoritos) {

    localStorage.setItem(
        STORAGE.favorites,
        JSON.stringify(favoritos)
    );
}


function esFavorito(id) {

    return cargarFavoritos().includes(id);
}


function guardarFavorito(id) {

    let favoritos = cargarFavoritos();

    if (!favoritos.includes(id)) {

        favoritos.push(id);

    } else {

        favoritos =
            favoritos.filter(item => item !== id);
    }

function buscarIA(query) {

    const q = String(query || "")
        .toLowerCase()
        .trim();

    if (!q) {

        return obtenerTodasLasIA();
    }

    return obtenerTodasLasIA().filter(ia => {

        const contenido = [

            ia.nombre,
            ia.empresa,
            ia.descripcion,
            ...(ia.categorias || []),
            ...(ia.especialidades || []),
            ...(ia.tags || [])

        ]
        .join(" ")
        .toLowerCase();

        return contenido.includes(q);
    });
}


/* ============================================================
   FILTROS
   ============================================================ */

function filtrarIA(lista, filtros = {}) {

    return lista.filter(ia => {

        if (
            filtros.precio &&
            ia.precio !== filtros.precio
        ) {
            return false;
        }

        if (
            filtros.categoria &&
            !ia.categorias.includes(filtros.categoria)
        ) {
            return false;
        }

        if (
            filtros.gratuita !== undefined &&
            ia.gratuita !== filtros.gratuita
        ) {
            return false;
        }

        if (
            filtros.api !== undefined &&
            ia.api !== filtros.api
        ) {
            return false;
        }

        if (filtros.funcion) {

            const funcion =
                filtros.funcion.toLowerCase();

            const existe =
                [
                    ...(ia.tags || []),
                    ...(ia.especialidades || [])
                ]
                .join(" ")
                .toLowerCase()
                .includes(funcion);

            if (!existe) {

                return false;
            }
        }

        return true;
    });
}


/* ============================================================
   ORDENAMIENTO
   ============================================================ */

function ordenarIA(lista, criterio = "score") {

    const copia = [...lista];

    switch (criterio) {

        case "az":

            return copia.sort(
                (a, b) =>
                    a.nombre.localeCompare(b.nombre)
            );

        case "za":

            return copia.sort(
                (a, b) =>
                    b.nombre.localeCompare(a.nombre)
            );

        case "popularidad":

            return copia.sort(
                (a, b) =>
                    b.popularidad - a.popularidad
            );

        case "potencia":

            return copia.sort(
                (a, b) =>
                    b.potencia - a.potencia
            );

        case "calidad":

            return copia.sort(
                (a, b) =>
                    b.calidad - a.calidad
            );

        case "fecha":

            return copia.sort(
                (a, b) =>
                    b.fechaActualizacion
                    .localeCompare(a.fechaActualizacion)
            );

        default:

            return copia.sort(
                (a, b) =>
                    b.score - a.score
            );
    }
}

    ]

function calcularMatch(ia, consulta, opciones = {}) {

    const q = String(consulta || "")
        .toLowerCase();

    if (!q.trim()) {

        return 0;
    }

    let score = 0;

    const palabras =
        q
            .split(/[^a-záéíóúüñ0-9]+/)
            .filter(Boolean);

    const contenido = [

        ia.nombre,
        ia.empresa,
        ia.descripcion,
        ...(ia.categorias || []),
        ...(ia.especialidades || []),
        ...(ia.tags || [])

    ]
    .join(" ")
    .toLowerCase();


    palabras.forEach(palabra => {

        if (contenido.includes(palabra)) {

            score += 12;
        }
    });


    Object.entries(KEYWORDS)
        .forEach(([categoria, palabrasClave]) => {

            const detectada =
                palabrasClave.some(
                    palabra =>
                        q.includes(palabra)
                );

            if (!detectada) {
                return;
            }

            const coincide =
                ia.categorias.some(
                    cat =>
                        cat
                            .toLowerCase()
                            .includes(categoria)
                );

            if (coincide) {

                score += 30;
            }
        });


    if (
        opciones.precio &&
        ia.precio === opciones.precio
    ) {

        score += 15;
    }


    if (opciones.prioridad === "Potencia") {

        score += ia.potencia / 10;
    }

    if (opciones.prioridad === "Calidad") {

        score += ia.calidad / 10;
    }

    if (opciones.prioridad === "Facilidad") {

        score += ia.facilidad / 10;
    }

    if (
        opciones.prioridad === "Precio" &&
        ia.gratuita
    ) {

        score += 10;
    }


    return Math.min(
        99,
        Math.max(
            35,
            Math.round(
                Math.min(100, score)
            )
        )
    );
}


/* ============================================================
   RECOMENDADOR
   ============================================================ */

function obtenerRecomendaciones(
    consulta,
    opciones = {}
) {

    return obtenerTodasLasIA()

        .map(ia => ({

            ...ia,

            match:
                calcularMatch(
                    ia,
                    consulta,
                    opciones
                )

        }))

        .sort(
            (a, b) =>
                b.match - a.match ||
                b.score - a.score
        )

        .slice(0, 5);
}

                    ${

                            ia => `

                                <td
                                    class="${
                                        ia[criterio] === max
                                            ? "best"
                                            : ""
                                    }"
                                >

                                    ${ia[criterio]}

                                    ${
                                        ia[criterio] === max
                                            ? " ✓"
                                            : ""
                                    }

                                </td>

                            `
                        ).join("")
                    }

                </tr>

            `;
        }
    );


    html += `

            </tbody>

        </table>

        <p class="muted">

            El mejor resultado de cada
            fila se resalta automáticamente.

        </p>

    `;


    const modalBody =
        $("#modalBody");

    if (modalBody) {

        modalBody.innerHTML = html;
    }


    $("#modal")
        ?.classList
        .add("open");
}


/* ============================================================
   CATÁLOGO
   ============================================================ */

function aplicarFiltros() {

    const search =
        $("#globalSearch")
            ?.value
            .toLowerCase()
            .trim() || "";


    const precio =
        $("#priceFilter")
            ?.value || "";


    const categoria =
        $("#categoryFilter")
            ?.value || "";


    const funcion =
        $("#functionFilter")
            ?.value || "";


    const orden =
        $("#sortFilter")
            ?.value || "score";


    let lista =
        obtenerTodasLasIA();


    if (search) {

        lista =
            buscarIA(search);
    }


    lista =
        filtrarIA(
            lista,
            {
                precio,
                categoria,
                funcion
            }
        );


    lista =
        ordenarIA(
            lista,
            orden
        );


    currentResults =
        lista;

    currentPage = 1;

    renderizarCatalogo();
}


function renderizarCatalogo() {

    const inicio =
        (currentPage - 1) *
        PAGE_SIZE;


    const fin =
        inicio + PAGE_SIZE;


    const resultados =
        currentResults.slice(
            inicio,
            fin
        );


    renderizarIA(
        "catalogGrid",
        resultados
    );


    const totalPaginas =
        Math.max(
            1,
            Math.ceil(
                currentResults.length /
                PAGE_SIZE
            )
        );


    const contador =
        $("#resultCount");

    if (contador) {

        contador.textContent = `

            ${currentResults.length}
            resultados · página
            ${currentResults.length ? currentPage : 0}
            de
            ${totalPaginas}

        `;
    }


    renderizarPaginacion(
        totalPaginas
    );
}


/* ============================================================
   PAGINACIÓN
   ============================================================ */

function renderizarPaginacion(
    totalPaginas
) {

    const elemento =
        $("#pagination");

    if (!elemento) {
        return;
    }


    let html = "";


    for (
        let i = 1;
        i <= totalPaginas;
        i++
    ) {

        html += `

            <button
                class="pagebtn ${
                    i === currentPage
                        ? "active"
                        : ""
                }"
                onclick="irPagina(${i})"
            >

                ${i}

            </button>

        `;
    }


    elemento.innerHTML =
        html;
}


function irPagina(numero) {

    currentPage =
        numero;

    renderizarCatalogo();


    $("#explorar")
        ?.scrollIntoView({
            behavior: "smooth",
            block: "start"
        });
}

function exportarCatalogo() {

    const datos =
        obtenerTodasLasIA();


    const blob =
        new Blob(
            [
                JSON.stringify(
                    datos,
                    null,
                    2
                )
            ],
            {
                type:
                    "application/json"
            }
        );


    const url =
        URL.createObjectURL(
            blob
        );


    const enlace =
        document.createElement("a");


    enlace.href =
        url;

    enlace.download =
        "ai-world-hub.json";


    enlace.click();


    URL.revokeObjectURL(
        url
    );
}


async function importarCatalogo(
    archivo
) {

    if (!archivo) {

        throw new Error(
            "No se seleccionó ningún archivo."
        );
    }


    const texto =
        await archivo.text();


    const datos =
        JSON.parse(texto);


    if (!Array.isArray(datos)) {

        throw new Error(
            "El JSON debe contener un array."
        );
    }


    const existentes =
        loadCustomIA();


    const nuevos =
        datos.filter(
            item =>
                item.id &&
                !existentes.some(
                    existente =>
                        existente.id === item.id
                )
        );


    saveCustomIA(
        [
            ...existentes,
            ...nuevos
        ]
    );


    customIA =
        loadCustomIA();


    renderAll();


    return nuevos.length;
}

function obtenerIAAleatoria() {

    const todas =
        obtenerTodasLasIA();

    if (!todas.length) {
        return null;
    }


    const indice =
        Math.floor(
            Math.random() *
            todas.length
        );


    return todas[indice];
}
